local function setNoclip(state)
    if noclipActive == state or not character then return end
    noclipActive = state
    for _, part in pairs(character:GetDescendants()) do
        if part:IsA("BasePart") then part.CanCollide = not state end
    end
end

local function startAntiAim()
    if AntiAimConnection then return end
    AntiAimConnection = RS.Heartbeat:Connect(function()
        if not getgenv().AntiAimAssiant or antiAimCooldown then return end
        local hum = Char.get_hum()
        if hum and not hum:GetAttribute("HasBeenDowned") then
            local root = Char.get_hrp()
            if root then
                antiAimCooldown = true
                local A, B, C = root.Velocity, root.AssemblyLinearVelocity, root.AssemblyAngularVelocity
                root.Velocity = Vector3.new(math.random(-5000, 5000), math.random(-5000, 5000), math.random(-5000, 5000))
                root.AssemblyLinearVelocity = Vector3.new(math.random(-5000, 5000), math.random(-5000, 5000), math.random(-5000, 5000))
                root.AssemblyAngularVelocity = Vector3.new(math.random(-5000, 5000), math.random(-5000, 5000), math.random(-5000, 5000))
                RS.RenderStepped:Wait()
                root.Velocity, root.AssemblyLinearVelocity, root.AssemblyAngularVelocity = A, B, C
                task.wait(0.05)
                antiAimCooldown = false
            end
        end
    end)
end

local function stopAntiAim()
    if AntiAimConnection then AntiAimConnection:Disconnect(); AntiAimConnection = nil end
end

        isEvading = false

        if originalPos and rootPart then rootPart.CFrame = originalPos end
        if getgenv().AntiAimAssiant then startAntiAim() end
    end

-- Helper functions
local function GetPlayerNames()
    local t = {}
    for _, v in ipairs(Plr:GetPlayers()) do
        if v ~= player then table.insert(t, v.Name) end
    end
    return t
end

local function WorldToViewPoint(pos)
    return Camera:WorldToViewportPoint(pos)
end

local function IsAlive(model)
    local hum = model:FindFirstChildOfClass("Humanoid")
    local root = model:FindFirstChild("HumanoidRootPart")
    return hum and root and hum.Health > 0
end

local function GetVelocity(target, pos)
    local t = tick()
    TargetHistory[target] = TargetHistory[target] or {}
    local hist = TargetHistory[target]
    if #hist >= 3 then table.remove(hist, 1) end
    table.insert(hist, { pos = pos, time = t })
    if #hist < 2 then return Vector3.zero end
    local p1, p2 = hist[#hist - 1], hist[#hist]
    local dt = math.max(p2.time - p1.time, 1e-6)
    return (p2.pos - p1.pos) / dt
end

local function GetClosestTarget()
    local closest, closestDist = nil, math.huge
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    
    for _, v in ipairs(Plr:GetPlayers()) do
        if v ~= player and not v:GetAttribute("SilentAimIgnore") then
            local char = v.Character
            if char and IsAlive(char) then
                local root = char:FindFirstChild("HumanoidRootPart")
                if root then
                    local pos, onScreen = WorldToViewPoint(root.Position)
                    if onScreen then
                        local d = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                        if d < FOV and d < closestDist then
                            closest, closestDist = char, d
                        end
                    end
                end
            end
        end
    end
    return closest
end

local function DebugGun(fromPos, toPos)
    local distance = (toPos - fromPos).Magnitude
    local part = Instance.new("Part")
    part.Size = Vector3.new(0.6, 0.6, distance)
    part.CFrame = CFrame.new(fromPos, toPos) * CFrame.new(0, 0, -distance/2)
    part.Anchored = true
    part.CanCollide = false
    part.Material = Enum.Material.Neon
    part.Color = Color3.fromRGB(255, 255, 255)
    part.Parent = workspace
    Deb:AddItem(part, 1)
end

-- Hook shoot_gun
local OldSend = hookfunction(Network.send, function(...)
    local args = { ... }
    if args[1] == "shoot_gun" and SilentAimEnabled then
        local target = GetClosestTarget()
        if target then
            local plr = Plr:GetPlayerFromCharacter(target)
            if plr and plr:GetAttribute("SilentAimIgnore") then
                return OldSend(...)
            end
            local part = target:FindFirstChild(TargetPartName)
            if part then
                local char = player.Character
                local root = char and char:FindFirstChild("HumanoidRootPart")
                if root then
                    local myPos, targetPos = root.Position, part.Position
                    local vel = GetVelocity(target, targetPos)
                    local predictedPos = vel.Magnitude >= HIGH_VEL_THRESHOLD and targetPos or targetPos
                    args[3] = CFrame.new(myPos, predictedPos)
                    if DebugEnabled then DebugGun(myPos, predictedPos) end
                    for _, v in pairs(args[4] or {}) do
                        for _, x in pairs(v) do
                            x.Position = predictedPos
                            x.Instance = part
                        end
                    end
                end
            end
        end
    end
    return OldSend(table.unpack(args))
end)

                    local width, height = math.abs(scale) * 0.6, math.abs(scale)
                    ESPObjects.Boxes[p].Size = Vector2.new(width, height)
                    ESPObjects.Boxes[p].Position = Vector2.new(pos.X - width / 2, pos.Y - height / 2)
                    ESPObjects.Boxes[p].Visible = true
                elseif ESPObjects.Boxes[p] then
                    ESPObjects.Boxes[p].Visible = false
                end
                
                -- ESP Name
                if NameESPEnabled and ESPObjects.Names[p] and headPos and headPos.Z > 0 then
                    ESPObjects.Names[p].Text = p.Name
                    ESPObjects.Names[p].Position = Vector2.new(headPos.X, headPos.Y - 20)
                    ESPObjects.Names[p].Visible = true
                elseif ESPObjects.Names[p] then
                    ESPObjects.Names[p].Visible = false
                end
                
                -- ESP Health
                if HealthESPEnabled and ESPObjects.HealthBars[p] and visible then
                    local top = Camera:WorldToViewportPoint(hrp.Position + Vector3.new(0, 3, 0))
                    local bottom = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
                    local height = math.abs(top.Y - bottom.Y)
                    local healthPercent = hum.Health / hum.MaxHealth
                    local barHeight = height * healthPercent
                    local x = pos.X - (height * 0.35)
                    ESPObjects.HealthBars[p].From = Vector2.new(x, bottom.Y)
                    ESPObjects.HealthBars[p].To = Vector2.new(x, bottom.Y - barHeight)
                    ESPObjects.HealthBars[p].Color = Color3.fromRGB(255 - (255 * healthPercent), 255 * healthPercent, 0)
                    ESPObjects.HealthBars[p].Visible = true
                elseif ESPObjects.HealthBars[p] then
                    ESPObjects.HealthBars[p].Visible = false
                end
                
                -- ESP Distance
                if DistanceESPEnabled and ESPObjects.Distances[p] and myPos and visible then
                    local distance = math.floor((myPos - hrp.Position).Magnitude)
                    ESPObjects.Distances[p].Text = distance .. "m"
                    ESPObjects.Distances[p].Position = Vector2.new(pos.X, pos.Y + 30)
                    ESPObjects.Distances[p].Visible = true
                elseif ESPObjects.Distances[p] then
                    ESPObjects.Distances[p].Visible = false
                end
                
                -- ESP Tracer
                if TracerEnabled and ESPObjects.Tracers[p] and visible then
                    ESPObjects.Tracers[p].From = Vector2.new(center.X, 0)
                    ESPObjects.Tracers[p].To = Vector2.new(pos.X, pos.Y)
                    ESPObjects.Tracers[p].Visible = true
                elseif ESPObjects.Tracers[p] then
                    ESPObjects.Tracers[p].Visible = false
                end
            else
                if ESPObjects.Boxes[p] then ESPObjects.Boxes[p].Visible = false end
                if ESPObjects.Names[p] then ESPObjects.Names[p].Visible = false end
                if ESPObjects.HealthBars[p] then ESPObjects.HealthBars[p].Visible = false end
                if ESPObjects.Distances[p] then ESPObjects.Distances[p].Visible = false end
                if ESPObjects.Tracers[p] then ESPObjects.Tracers[p].Visible = false end
            end
            
            UpdateHighlight(p, HighlightESPEnabled)
        end
    end

local BlueColor = Color3.fromRGB(0, 150, 255)

local function getItemColor(item)
    return item.Name:lower():find("money") and GreenColor or BlueColor
end

local function createItemESP(item)
    if ItemESPs[item] then return end
    local color = getItemColor(item)
    local highlights = {}
    
    local function addHighlight(part)
        local hl = Instance.new("Highlight")
        hl.Adornee = part
        hl.FillColor = color
        hl.OutlineColor = color
        hl.FillTransparency = 0.8
        hl.OutlineTransparency = 0
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.Enabled = true
        hl.Parent = part
        table.insert(highlights, hl)
    end
    
    if item:IsA("BasePart") then
        addHighlight(item)
    elseif item:IsA("Model") then
        for _, v in ipairs(item:GetDescendants()) do
            if v:IsA("BasePart") then addHighlight(v) end
        end
    end
    
    local basePart = item:IsA("BasePart") and item or item.PrimaryPart or item:FindFirstChildWhichIsA("BasePart")
    local label = nil
    
    if basePart then
        local bb = Instance.new("BillboardGui")
        bb.Adornee = basePart
        bb.Size = UDim2.new(0, 55, 0, 9)
        bb.StudsOffset = Vector3.new(0, basePart.Size.Y / 2 + 0.6, 0)
        bb.AlwaysOnTop = true
        bb.Parent = basePart
        label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, 0, 1, 0)
        label.BackgroundTransparency = 1
        label.Text = "[" .. item.Name .. "]"
        label.Font = Enum.Font.GothamBold
        label.TextSize = 8
        label.TextColor3 = color
        label.TextStrokeTransparency = 0.5
        label.Visible = true
        label.Parent = bb
    end
    
    ItemESPs[item] = { highlights = highlights, label = label }
end

local function removeItemESP(item)
    local data = ItemESPs[item]
    if not data then return end
    for _, hl in ipairs(data.highlights) do if hl then hl:Destroy() end end
    if data.label and data.label.Parent then data.label.Parent:Destroy() end
    ItemESPs[item] = nil
end

for _, item in ipairs(DroppedFolder:GetChildren()) do createItemESP(item) end
DroppedFolder.ChildAdded:Connect(createItemESP)
DroppedFolder.ChildRemoved:Connect(removeItemESP)

        Camera.CameraType = Enum.CameraType.Custom

end
